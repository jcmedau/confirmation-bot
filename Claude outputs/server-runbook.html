01

Create Docker Network

All containers communicate over this bridge network. Run once — skip if it already exists.

bash
sudo docker network create mysql-network
verify
sudo docker network ls | grep mysql-network
02

Start Redis

Used by Evolution API as its cache provider. No persistent volume needed — cache is ephemeral.

bash
sudo docker run -d \
  --name redis \
  --network mysql-network \
  --restart unless-stopped \
  redis:latest
verify
sudo docker logs redis --tail 5
# Expected: Ready to accept connections
03

Start PostgreSQL

Evolution API v2 requires PostgreSQL — it uses Prisma with PostgreSQL-dialect raw queries even when configured for another provider.

bash
sudo docker run -d \
  --name postgres \
  --network mysql-network \
  --restart unless-stopped \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=SkyPowerDat@tabase2026 \
  -e POSTGRES_DB=evolution \
  -v postgres_data:/var/lib/postgresql/data \
  postgres:15
verify
sudo docker logs postgres --tail 5
# Expected: database system is ready to accept connections
04

Start Evolution API

WhatsApp gateway. Connects to PostgreSQL for state and Redis for cache. Wait for PostgreSQL to be fully ready before running this.

bash
sudo docker run -d \
  --name evolution_api \
  --network mysql-network \
  -p 8083:8080 \
  --restart unless-stopped \
  -v evolution_instances:/evolution/instances \
  -e ENV=prod \
  -e DATABASE_CONNECTION_URI="postgresql://postgres:SkyPowerDat%40tabase2026@postgres:5432/evolution" \
  -e DATABASE_ENABLED=true \
  -e DATABASE_PROVIDER=postgresql \
  -e CACHE_PROVIDER=redis \
  -e CACHE_REDIS_URI=redis://redis:6379 \
  -e CACHE_ENABLED=true \
  -e AUTHENTICATION_TYPE=apikey \
  -e AUTHENTICATION_API_KEY=SkyPowerWapp2026! \
  -e SERVER_URL=http://messages.skypower-aviation.com \
  -e CONFIG_SESSION_PHONE_CLIENT=Evolution \
  -e WEBSOCKET_ENABLED=true \
  evoapicloud/evolution-api:latest
verify
curl -s http://localhost:8083/ -H 'apikey: SkyPowerWapp2026!'
# Expected: JSON response (not connection refused)
05

Start Appointment Bot

Build the jar on your Mac, copy it to the server, then build and run the Docker image.

A — Build on Mac
mac terminal
cd ~/Developer/confirmation-bot
./mvnw clean package -DskipTests
B — Copy jar to server
mac terminal
scp target/confirmation-bot-*.jar medau@skypowerserver:~/confirmation-bot.jar
scp Dockerfile medau@skypowerserver:~/Dockerfile
scp credentials/service-account.json medau@skypowerserver:~/service-account.json
C — Build image on server
server
cd ~
sudo docker build --no-cache -t appointment-bot .
D — Run container
server
sudo docker run -d \
  --name appointment-bot \
  --network mysql-network \
  -p 8084:8080 \
  --restart unless-stopped \
  appointment-bot
verify
sudo docker logs appointment-bot --tail 20
# Expected: Started ConfirmationBotApplication
06

Configure WhatsApp Instance

Only needed when setting up from scratch or after the WhatsApp session was removed as a linked device. If the evolution_instances volume is intact from a previous run, the session persists — skip to step 7.
A — Create instance
bash
curl -s -X POST 'http://localhost:8083/instance/create' \
  -H 'apikey: SkyPowerWapp2026!' \
  -H 'Content-Type: application/json' \
  -d '{"instanceName": "skypower-main", "qrcode": true}'
B — Get QR code
bash
curl -s 'http://localhost:8083/instance/connect/skypower-main' \
  -H 'apikey: SkyPowerWapp2026!'
# Returns a base64 QR code. Open WhatsApp → Linked Devices → scan it.
C — Verify connection
bash
curl -s 'http://localhost:8083/instance/connectionState/skypower-main' \
  -H 'apikey: SkyPowerWapp2026!'
# Expected: {"instance":{"instanceName":"skypower-main","state":"open"}}
07

Register Webhook

Tells Evolution API to call the appointment bot whenever a message is received. The bot also auto-registers this on startup via EvolutionWebhookRegistrar — run this manually only if the bot isn't doing it or you need to verify.

bash
curl -s -X POST 'http://localhost:8083/webhook/set/skypower-main' \
  -H 'apikey: SkyPowerWapp2026!' \
  -H 'Content-Type: application/json' \
  -d '{
    "url": "http://appointment-bot:8080/webhook/evolution",
    "webhook_by_events": false,
    "webhook_base64": false,
    "events": ["MESSAGES_UPSERT"]
  }'
Verify webhook is registered
bash
curl -s 'http://localhost:8083/webhook/find/skypower-main' \
  -H 'apikey: SkyPowerWapp2026!'
# Expected: shows url "http://appointment-bot:8080/webhook/evolution"

Day-to-Day Commands

Check all containers
bash
sudo docker ps --format 'table {{.Names}}\t{{.Status}}\t{{.Ports}}'
Follow bot logs
bash
sudo docker logs appointment-bot -f --tail 50
Follow Evolution API logs
bash
sudo docker logs evolution_api -f --tail 50
Redeploy bot only (new jar built)
bash
sudo docker stop appointment-bot && sudo docker rm appointment-bot
sudo docker build --no-cache -t appointment-bot .
sudo docker run -d \
  --name appointment-bot \
  --network mysql-network \
  -p 8084:8080 \
  --restart unless-stopped \
  appointment-bot
Simulate incoming patient reply (test webhook)
bash
curl -s -X POST 'http://localhost:8084/webhook/evolution' \
  -H 'Content-Type: application/json' \
  -d '{
    "event": "messages.upsert",
    "instance": "skypower-main",
    "data": {
      "key": {
        "remoteJid": "5519991278903@s.whatsapp.net",
        "fromMe": false,
        "id": "TEST001"
      },
      "message": { "conversation": "1" },
      "pushName": "Paciente Teste"
    }
  }'
Restart all services
bash
sudo docker restart redis postgres evolution_api appointment-bot
Wipe and recreate everything (nuclear reset)
⚠ This deletes all WhatsApp session data. You will need to scan the QR code again.
bash
sudo docker stop appointment-bot evolution_api postgres redis
sudo docker rm appointment-bot evolution_api postgres redis
sudo docker volume rm evolution_instances postgres_data
# Then re-run steps 1–7 above.
